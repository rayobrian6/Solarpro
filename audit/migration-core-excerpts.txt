  mounting_id           VARCHAR(100),
  mount_type            VARCHAR(50),

  -- Electrical

  conduit_type          VARCHAR(30),
  rapid_shutdown        BOOLEAN        DEFAULT true,

  dc_disconnect         BOOLEAN        DEFAULT true,

  -- Site

  utility_id            VARCHAR(100),
  state_code            VARCHAR(2),
  address               TEXT,
  ahj                   VARCHAR(200),
  roof_pitch            INTEGER,
